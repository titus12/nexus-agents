RAW = "帮我审核当前项目未提交的代码，生成可执行的审核方案"
BASELINE = [{"requirement_id": "req-000001", "statement": RAW, "source_quote": RAW,
             "priority": "must", "acceptance_signal": "审核方案可以按步骤执行并输出风险报告"}]


def draft(count=2, interleaved=False):
    items = [{"item_id": f"item-{n:06d}", "group_id": "group-000001", "title": f"审核步骤 {n}",
              "objective": f"产出审核步骤 {n} 的检查报告", "dependencies": [] if n == 1 else [f"item-{n-1:06d}"],
              "source_requirement_ids": ["req-000001"], "acceptance_signals": [f"步骤 {n} 的风险报告可读取"]}
             for n in range(1, count + 1)]
    groups = [{"group_id": "group-000001", "title": "代码审核", "objective": "产生审核方案", "item_ids": [i["item_id"] for i in items]}]
    if interleaved:
        for n, i in enumerate(items):
            i["group_id"] = f"group-{(n % 2) + 1:06d}"
        groups = [{"group_id": f"group-{n:06d}", "title": f"业务组 {n}", "objective": "审核方案",
                   "item_ids": [i["item_id"] for i in items if i["group_id"] == f"group-{n:06d}"]} for n in (1, 2)]
    return {"requirements": copy.deepcopy(BASELINE), "items": items, "groups": groups,
            "scope": "输出审核方案；实际代码检查待执行", "unknowns": []}


def executable(plan):
    plan = copy.deepcopy(plan)
    for i in plan["items"]:
        i.update(inputs=[{"name": "审核需求", "kind": "requirement", "source_id": "req-000001", "output_name": ""}],
                 steps=["读取 diff 并记录涉及模块", "检查变更并记录证据及风险"],
                 outputs=[{"name": "risk-report", "description": "变更风险与证据列表"}],
                 verification=[{"method": "检查报告中的结论是否均有代码位置或测试证据", "expected_result": "风险报告完整且可追溯", "status": "planned"}],
                 risks="缺少运行环境时记录未验证项", compatibility="只生成方案，不改变当前项目接口", rollback="不适用：本工作项是只读审核")
    plan["decisions"] = []
    return plan


def context():
    return build_context(Namespace(project="example-project", project_type="python", task_type="planning", plan_schema=None, workflow=None),
                         "issue-planning", RAW, "task-planning-test")


class ScriptedPlanning(FakeMulticaAdapter):
    def __init__(self, mutate=None, plan=None):
        super().__init__()
        self.mutate = mutate
        self.plan = plan or draft()
        self.counts = Counter()

    def dispatch(self, request):
        receipt = super().dispatch(request)
        state = request.target_state
        self.counts[state] += 1
        payload = {"action": {"PLANNING_BASELINE": "BASELINE_READY", "PLANNING_EVIDENCE": "EVIDENCE_READY",
                              "PLANNING_DRAFT": "DRAFT_READY", "PLANNING_EXECUTION_DRAFT": "EXECUTION_READY",
                              "PLANNING_EXECUTION_REVIEW": "REVIEW_APPROVED"}[state], "questions": []}
        if state == "PLANNING_BASELINE":
            payload["requirements"] = copy.deepcopy(BASELINE)
        elif state == "PLANNING_EVIDENCE":
            payload["evidence"] = []
        elif state == "PLANNING_DRAFT":
            payload["plan"] = copy.deepcopy(self.plan)
        elif state == "PLANNING_EXECUTION_DRAFT":
            payload["plan"] = executable(request.context["plan"])
        else:
            payload.update(reviewed_revision_ref=request.context["plan_revision_ref"], issues=[])
        if self.mutate:
            self.mutate(self, request, payload)
        payload.update(task_id=request.task_id, request_id=request.request_id, phase=request.phase, role=request.role,
                       contract_id=request.context["contract_id"], state=state, mode="PLAN")
        self.queue_reply(request.request_id, ExternalMessage(request.agent_id, payload))
        return receipt


class PlanningApplicationTests(unittest.TestCase):
    def step(self, app):
        task = app.context.identity.task_id
        event = app.inbox.next(task)
        if event:
            app.engine.dispatch(event)
            app.inbox.ack(event)
        elif not app.repository.pending_effects(task):
            self.fail("No work in " + app.ctx.progression.state)
        app.effects.execute_pending(task)

    def drive(self, app, *, approve=True):
        app._ensure_start_event()
        for _ in range(150):
            c = app.ctx
            if c.progression.state in TERMINAL_STATES:
                return c
            if c.progression.state == "HUMAN_GATE":
                if not approve or c.human_gate.reason_code != "PLAN_REVIEW":
                    return c
                self.assertTrue(app.resume("通过"))
            self.step(app)
        self.fail("Planning failed to finish in a bounded number of steps")

    def app(self, directory, adapter):
        return OrchestratorApp(context(), root=directory, multica=adapter, poll_interval=0, timeout_seconds=2)

    def test_default_recipe_publishes_verified_revision_and_planned_work(self):
        with tempfile.TemporaryDirectory() as directory:
            adapter = ScriptedPlanning()
            app = self.app(directory, adapter)
            final = self.drive(app)
            self.assertEqual(final.progression.state, "PLAN_READY", final.planning.reason)
            self.assertEqual(final.planning.calls, 5)
            self.assertEqual(final.review.plan_revision.number, 2)
            self.assertEqual(final.planning.reviewed_revision_ref, final.review.plan_revision.reference)
            self.assertNotEqual(final.planning.requirements_approval_ref, final.planning.reviewed_revision_ref)
            path = Path(final.planning.result_ref)
            self.assertTrue(path.is_file())
            self.assertIn("验证均为待执行", path.read_text(encoding="utf-8"))
            report = json.loads(path.with_name("validation.json").read_text(encoding="utf-8"))
            self.assertEqual(report["work_verification_status"], "planned")
            saved = context_from_dto(context_to_dto(final))
            self.assertEqual(saved.planning, final.planning)
            self.assertTrue(app.run())
            from orchestrator.notifications import build_notification
            from orchestrator.domain.events import DomainEvent
            notice = build_notification("PLAN_READY", "DONE", DomainEvent("PUBLISHED", final.identity.task_id,
                                        final.progression.sequence, {}, "now"), final)
            self.assertIn(final.planning.result_ref, notice)
            self.assertIn("验证仍待执行", notice)
            self.assertTrue(all(r.phase == "PLANNING" and r.target_state.startswith("PLANNING_") for r in adapter.dispatched))

    def test_large_business_group_and_interleaved_labels_are_schedulable(self):
        for plan in (draft(8), draft(3, True)):
            with self.subTest(groups=plan["groups"]), tempfile.TemporaryDirectory() as directory:
                final = self.drive(self.app(directory, ScriptedPlanning(plan=plan)))
                self.assertEqual(final.progression.state, "PLAN_READY", final.planning.reason)
                self.assertEqual(len(execution_order(final.review.plan_revision.plan)), len(plan["items"]))
                with self.assertRaises(PlanCompileError):
                    compile_plan(plan)

    def test_human_requirement_rejection_and_restart_require_fresh_approval(self):
        def mutate(adapter, request, reply):
            if request.target_state == "PLANNING_DRAFT" and adapter.counts[request.target_state] == 2:
                reply["plan"]["items"][0]["objective"] = "按用户反馈细化审核范围"
        with tempfile.TemporaryDirectory() as directory:
            adapter = ScriptedPlanning(mutate)
            app = self.app(directory, adapter)
            first = self.drive(app, approve=False)
            self.assertEqual(first.human_gate.reason_code, "PLAN_REVIEW")
            self.assertFalse(app.resume())
            self.assertTrue(app.resume("拒绝：细化审核范围"))
            self.step(app)
            second = self.drive(app, approve=False)
            self.assertEqual(second.review.plan_revision.number, 2)
            self.assertNotEqual(second.human_gate.document_hash, first.human_gate.document_hash)
            restored = OrchestratorApp(first, root=directory, multica=adapter, poll_interval=0, timeout_seconds=2)
            self.assertEqual(restored.ctx.review.plan_revision, second.review.plan_revision)
            final = self.drive(restored)
            self.assertEqual(final.progression.state, "PLAN_READY", final.planning.reason)

    def test_success_publication_is_replayable_and_changed_bytes_are_rejected(self):

class PlanningPolicyTests(unittest.TestCase):
    def test_production_reply_contract_and_schema_follow_each_exact_stage(self):
        from orchestrator.contracts.planning import CONTRACTS
        from orchestrator.structured_output import build_structured_output_spec, role_result_template
        from orchestrator.adapters import _response_contract_for
        from orchestrator.transport.external import AgentRequest
        for active in CONTRACTS:
            with self.subTest(state=active.state):
                dispatch_context = {"active_runtime_state": active.state, "target_state": active.state}
                spec = build_structured_output_spec(active.phase, active.role, dispatch_context)
                self.assertEqual(spec.state, active.state)
                request = AgentRequest("t", "r", "a", active.role, active.phase, "prompt", "key",
                                       context=dispatch_context, target_state=active.state, structured_output=spec.to_dict())
                response = _response_contract_for(request)
                self.assertEqual(response["allowed_actions"], list(active.actions))
                self.assertEqual(response["required"], list(active.required_fields))
                template = role_result_template(active.phase, active.role, state=active.state, schema_hash=spec.schema_hash,
                                                action=active.actions[0])
                self.assertEqual(active.validate(template), ())
                self.assertEqual(spec.schema["required"], response["required"])
                fields = spec.to_dict()["stable_fields"]
                if active.state == "PLANNING_EVIDENCE":
                    self.assertIn("evidence", fields)
                    self.assertNotIn("requirements", fields)

    def test_file_evidence_preserves_existing_verbatim_quote_gate(self):
        from orchestrator.domain.planning.execution import validate_evidence
        source = Path(__file__).resolve()
        row = {"evidence_id": "EV1", "source": str(source), "observation": "tests exist", "status": "observed"}
        with self.assertRaises(ValueError):
            validate_evidence([row])
        row["quote"] = {"path": str(source), "line_start": 1, "line_end": 1, "text": source.read_text(encoding="utf-8").splitlines()[0]}
        self.assertEqual(validate_evidence([row]), [row])
    def test_explicit_legacy_and_old_snapshots_keep_the_old_workflow(self):
        args = Namespace(project="p", project_type="python", task_type="review", plan_schema=None, workflow="legacy-v1")
        c = build_context(args, "i", RAW, "t")
        self.assertEqual((c.request.workflow, c.request.plan_schema), ("legacy-v1", "legacy-v1"))
        dto = context_to_dto(c)
        dto["request"].pop("workflow")
        dto.pop("planning")
        self.assertEqual(context_from_dto(dto).request.workflow, "legacy-v1")
        args.workflow = "planning-v1"
        args.plan_schema = "legacy-v1"
        with self.assertRaises(ValueError):
            build_context(args, "i", RAW, "t")

    def test_planning_graph_cannot_resume_into_legacy_states(self):
        with self.assertRaises(TransitionError):
            TransitionRegistry.planning().target_for("HUMAN_GATE", "RESUME", "MENXIA_GROUP_SOLVER")

    def test_cycle_and_unknown_inputs_still_fail_without_group_batch_limits(self):
        p = draft(8)
        self.assertEqual(len(compile_plan(p, policy_version="planning-v1")["items"]), 8)
        p["items"][0]["dependencies"] = [p["items"][-1]["item_id"]]
        with self.assertRaises(PlanCompileError):
            compile_plan(p, policy_version="planning-v1")
        p = executable(draft())
        p["items"][0]["inputs"][0].update(kind="evidence", source_id="invented")
        self.assertTrue(any(i["rule"] == "INPUT_UNBOUND" for i in execution_issues(p, [])))
        p["items"][0]["verification"][0]["status"] = "passed"
        self.assertTrue(any(i["rule"] == "VERIFICATION_PLANNED" for i in execution_issues(p, [])))
        p = draft()
        p["dependencies"] = [{"from": "item-000002", "to": "item-000001"}]
        with self.assertRaises(PlanCompileError):
            compile_plan(p, policy_version="planning-v1")
